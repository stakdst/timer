function formatTimer(ms) { ms = Math.max(0, ms); const totalHundredths = Math.floor(ms / 10); const hundredths = totalHundredths % 100; const totalSeconds = Math.floor(totalHundredths / 100); const seconds = totalSeconds % 60; const totalMinutes = Math.floor(totalSeconds / 60); const minutes = totalMinutes % 60; const hours = Math.floor(totalMinutes / 60); const ss = String(seconds).padStart(2, "0"); const mm = String(minutes).padStart(2, "0"); const hs = String(hundredths).padStart(2, "0"); // 1時間未満 if (hours === 0) { return `${mm}:${ss}.${hs}`; } // 1時間以上 // hourは2桁固定にせず、必要な桁数で表示 return `${hours}:${mm}:${ss}.${hs}`; }
